    sample_rate = 8000

    def fake_synthesizer(text, language, output_path):
        duration = 0.8 if text == "first" else 0.2
        time = np.arange(round(duration * sample_rate)) / sample_rate
        sf.write(output_path, 0.2 * np.sin(2 * np.pi * 440 * time), sample_rate)
        return True

    output_path = tmp_path / "timed.wav"
    synthesize_timed_speech(
        [
            {"start": 0.5, "end": 1.0, "text": "first"},
            {"start": 1.5, "end": 2.0, "text": "second"},
        ],
        "en",
        output_path,
        total_duration=2.5,
        synthesizer=fake_synthesizer,
        sample_rate=sample_rate,
    )

    audio, actual_rate = sf.read(output_path)
    assert actual_rate == sample_rate
    assert len(audio) == round(2.5 * sample_rate)
    assert np.max(np.abs(audio[:round(0.5 * sample_rate)])) == 0
    assert np.max(np.abs(audio[round(0.55 * sample_rate):round(0.95 * sample_rate)])) > 0.05
    assert np.max(np.abs(audio[round(1.05 * sample_rate):round(1.5 * sample_rate)])) == 0
    assert np.max(np.abs(audio[round(1.55 * sample_rate):round(1.95 * sample_rate)])) > 0.05
    assert np.max(np.abs(audio[round(2.0 * sample_rate):])) == 0

def _create_mux_inputs(
    tmp_path,
    *,
    video_duration=3.0,
    audio_duration=3.0,
    size="320x240",
    frame_rate="25",
    include_source_audio=False,
    subtitle_text="1\n00:00:00,500 --> 00:00:01,500\nañ¿añ«añ+aÑìaññaÑç añªaÑüañ¿añ+añ»añ+\n",
    video_filter=None,
    source_pixel_format="yuv420p",
):
    video_path = tmp_path / "input.mp4"
    audio_path = tmp_path / "dub.wav"
    subtitle_path = tmp_path / "translated.srt"
    output_path = tmp_path / "translated.mp4"

    video_command = [
        "ffmpeg", "-y", "-f", "lavfi", "-i",
        f"color=c=blue:s={size}:d={video_duration}:r={frame_rate}",
    ]
    if include_source_audio:
        video_command.extend([
            "-f", "lavfi", "-i", f"sine=frequency=220:duration={video_duration}",
            "-map", "0:v:0", "-map", "1:a:0", "-shortest",
        ])
    if video_filter:
        video_command.extend(["-vf", video_filter])
    video_command.extend([
        "-c:v", "libx264", "-pix_fmt", source_pixel_format, str(video_path),
    ])
    subprocess.run(video_command, check=True, capture_output=True)
    sample_rate = 44100
    time = np.arange(round(audio_duration * sample_rate)) / sample_rate
    sf.write(audio_path, 0.15 * np.sin(2 * np.pi * 440 * time), sample_rate)
    subtitle_path.write_text(subtitle_text, encoding="utf-8")
    return video_path, audio_path, subtitle_path, output_path


def _probe_streams(media_path):
    result = subprocess.run(
        ["ffprobe", "-v", "error", "-show_streams", "-of", "json", str(media_path)],
        check=True,
        capture_output=True,
        text=True,
    )
    return json.loads(result.stdout)["streams"]

    sf.write(input_path, audio, sample_rate)

    assert detect_dominant_voice_gender(
        input_path,

    ) == "male"

def test_mux_translated_video_contains_video_audio_and_subtitles(tmp_path):
    video_path, audio_path, subtitle_path, output_path = _create_mux_inputs(tmp_path)

    mux_translated_video(video_path, audio_path, subtitle_path, output_path, "hi")
    streams = _probe_streams(output_path)
    codecs = {
        (stream["codec_type"], stream["codec_name"])
        for stream in streams
    }
    subtitle_stream = next(
        stream
        for stream in streams
        if stream["codec_type"] == "subtitle"
    )

    assert [stream["codec_type"] for stream in streams] == ["video", "audio", "subtitle"]
    assert ("video", "h264") in codecs
    assert ("audio", "aac") in codecs
    assert ("subtitle", "mov_text") in codecs
    assert subtitle_stream.get("tags", {}).get("language") in {"hi", "hin"}
    assert subtitle_stream["disposition"]["default"] == 1
    assert probe_media_duration(output_path) == pytest.approx(3.0, abs=0.1)

    extracted_subtitles = subprocess.run(
        [
            "ffmpeg", "-v", "error", "-i", str(output_path), "-map", "0:s:0",
            "-f", "srt", "pipe:1",
        ],
        check=True,
        capture_output=True,
    ).stdout.decode("utf-8")
    assert "añ¿añ«añ+aÑìaññaÑç añªaÑüañ¿añ+añ»añ+" in extracted_subtitles

    frame = subprocess.run(
        [
            "ffmpeg", "-v", "error", "-ss", "0.5", "-i", str(output_path),
            "-frames:v", "1", "-f", "rawvideo", "-pix_fmt", "rgb24", "pipe:1",
        ],
        check=True,
        capture_output=True,
    )
    pixels = np.frombuffer(frame.stdout, dtype=np.uint8).reshape(240, 320, 3)
    attribution_region = pixels[8:55, 20:315]
    assert np.count_nonzero(np.all(attribution_region > 180, axis=2)) > 20

from datetime import datetime, timedelta, timezone
from statistics import median


IST = timezone(timedelta(hours=5, minutes=30), name="IST")
UTC = timezone.utc


def utc_to_ist(value: datetime | None) -> datetime | None:
    """Convert a stored naive UTC datetime to an IST-aware datetime."""
    if value is None:
        return None
    if value.tzinfo is None:
        value = value.replace(tzinfo=UTC)
    return value.astimezone(IST)


def format_ist(value: datetime | None, format_string: str) -> str:
    converted = utc_to_ist(value)
    return converted.strftime(format_string) if converted else "-"


def elapsed_seconds(started_at: datetime | None, ended_at: datetime | None = None) -> int | None:
    if started_at is None:
        return None
    ended_at = ended_at or datetime.utcnow()
    return max(0, round((ended_at - started_at).total_seconds()))


def estimate_remaining_seconds(
    elapsed: int | None,
    progress: int,
    historical_durations: list[int] | None = None,
) -> int | None:
    if elapsed is None or progress <= 0 or progress >= 100:
        return None
    progress_estimate = elapsed * (100 - progress) / progress
    historical_estimate = 0
    if historical_durations:
        historical_estimate = max(0, median(historical_durations) - elapsed)
    return max(1, round(max(progress_estimate, historical_estimate)))


def format_duration(seconds: int | float | None) -> str:
    if seconds is None:
        return "-"
    total_seconds = max(0, round(seconds))
    hours, remainder = divmod(total_seconds, 3600)
    minutes, seconds = divmod(remainder, 60)
    if hours:
        return f"{hours}h {minutes}m"
    if minutes:
        return f"{minutes}m {seconds}s"
    return f"{seconds}s"


def format_duration_between(started_at: datetime | None, ended_at: datetime | None) -> str:
    return format_duration(elapsed_seconds(started_at, ended_at))
